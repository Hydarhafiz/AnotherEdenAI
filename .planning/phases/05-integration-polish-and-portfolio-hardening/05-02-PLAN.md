**Test 4 — Retry cap exhaustion with mock (AGENT-05 + OUTPUT-04):**
```python
@pytest.mark.integration
async def test_retry_cap_exhaustion_returns_result(async_driver):
    """AGENT-05 + OUTPUT-04: After 3 VALIDATE failures, pipeline returns a dict — never a crash.

    Stubs validate_node to always return validation_errors without db_results.
    Confirms: exactly 3 validate calls occur, run.main() returns a dict,
    and the result is either an error schema or alternatives (not an unhandled exception).
    Uses async_driver (not loaded_db) — no real DB round-trips after stub.
    """
    from unittest.mock import patch

    call_count = 0

    async def always_fail_validate(state, driver):
        nonlocal call_count
        call_count += 1
        rc = state.get("retry_count", 0) + 1
        return {"validation_errors": [f"Stub fail attempt {rc}"], "retry_count": rc}

    with patch("src.workflow.nodes.validate.validate_node", side_effect=always_fail_validate):
        result = await run.main(["Aldo"], "test query for retry cap")

    assert call_count == 3, (
        f"Expected exactly 3 validate calls (retry cap = 3), got {call_count}. "
        "Check RETRY_CAP constant in validate.py."
    )
    assert isinstance(result, dict), (
        f"run.main() must return dict even after retry cap. Got: {type(result)}"
    )
```
