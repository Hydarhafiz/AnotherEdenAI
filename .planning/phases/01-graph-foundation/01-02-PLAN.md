```python
from pydantic import BaseModel, field_validator, model_validator
from typing import Optional
from .constants import STRICT

class CharacterRow(BaseModel):
    name: str
    element: str
    weapon: str
    light_shadow: str
    personalities: list[str]

    @field_validator("personalities", mode="before")
    @classmethod
    def parse_personalities(cls, v):
        if isinstance(v, str):
            return [p.strip() for p in v.split(",") if p.strip()]
        return v or []

class GrastaRow(BaseModel):
    name: str
    category: str
    tier: int
    stats: str
    personality_req: Optional[str] = None
    is_shareable: bool

    @field_validator("tier", mode="before")
    @classmethod
    def coerce_tier(cls, v):
        return int(v) if v is not None else 0

    @field_validator("is_shareable", mode="before")
    @classmethod
    def coerce_shareable(cls, v):
        if isinstance(v, str):
            return v == "1"
        return bool(v)

class OreRow(BaseModel):
    name: str
    stats: str
    source: str
```

```python
def parse_character(raw: dict) -> Optional[CharacterRow]:
    try:
        return CharacterRow.model_validate(raw, strict=STRICT)
    except Exception as exc:
        if STRICT:
            raise
        import logging
        logging.warning(f"Skipping character {raw.get('name')}: {exc}")
        return None

def parse_grasta(raw: dict) -> Optional[GrastaRow]:
    # Force personality_req=None for VC category
    if raw.get("category") == "VC":
        raw = {**raw, "personality_req": None}
    try:
        return GrastaRow.model_validate(raw, strict=STRICT)
    except Exception as exc:
        if STRICT:
            raise
        import logging
        logging.warning(f"Skipping grasta {raw.get('name')}: {exc}")
        return None

def parse_ore(raw: dict) -> Optional[OreRow]:
    try:
        return OreRow.model_validate(raw, strict=STRICT)
    except Exception as exc:
        if STRICT:
            raise
        import logging
        logging.warning(f"Skipping ore {raw.get('name')}: {exc}")
        return None
```

Create src/etl/loader.py using Pattern 3 from 01-RESEARCH.md:

ensure_constraints(driver) — creates 4 IF NOT EXISTS constraints (char_name, trait_name, grasta_name, ore_name).

load_characters(driver, rows: list[CharacterRow]) — UNWIND+MERGE for Character nodes; also MERGE Trait nodes and HAS_TRAIT edges per personality.

load_grastas(driver, rows: list[GrastaRow]) — UNWIND+MERGE for Grasta nodes (all categories); after Grasta nodes exist, create REQUIRES_TRAIT edges only where category != "VC" AND personality_req is not None:
```cypher
UNWIND $rows AS row
MATCH (g:Grasta {name: row.name})
WHERE row.category <> 'VC' AND row.personality_req IS NOT NULL AND row.personality_req <> ''
MERGE (t:Trait {name: row.personality_req})
MERGE (g)-[:REQUIRES_TRAIT]->(t)
```

load_ores(driver, rows: list[OreRow]) — UNWIND+MERGE for Ore nodes only. No ENHANCES edges created (wiki does not link Ore to specific Grasta names — documented in SCHEMA.md as deferred).
